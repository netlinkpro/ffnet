```c
// SPDX-License-Identifier: LGPL-2.1-or-later
#ifndef FFNET_INTERNAL_UTF8_H
#define FFNET_INTERNAL_UTF8_H

#include <stdint.h>

/* States returned by ff_utf8_decode. Only ACCEPT and REJECT are meaningful
 * to callers; other values indicate the decoder is mid-sequence. */
#define FF_UTF8_ACCEPT  0
#define FF_UTF8_REJECT 12

/* Feed one byte into the decoder. Caller starts a new message with
 *   state = FF_UTF8_ACCEPT
 * and feeds bytes one at a time. After each call:
 *   - state == FF_UTF8_REJECT → the byte sequence is not valid UTF-8;
 *     the caller should reject the message (no further bytes will recover).
 *   - state == FF_UTF8_ACCEPT → all bytes so far form valid complete UTF-8.
 *     This is the only state at which the message may end cleanly.
 *   - any other value → mid-sequence (a multi-byte codepoint is in
 *     progress and more bytes are needed).
 *
 * State carries across calls, so the decoder works incrementally across
 * fragmented WebSocket TEXT messages. */
uint32_t ff_utf8_decode(uint32_t state, uint8_t byte);

#endif /* FFNET_INTERNAL_UTF8_H */
```

```c
static uint32_t feed(uint32_t state, const char *bytes, size_t n)
{
    for (size_t i = 0; i < n; i++) {
        state = ff_utf8_decode(state, (uint8_t)bytes[i]);
        if (state == FF_UTF8_REJECT) return state;
    }
    return state;
}

static void test_utf8(void)
{
    /* Valid ASCII. */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "hello", 5), FF_UTF8_ACCEPT,
              "utf8 ASCII");

    /* Valid 2-byte (é = 0xC3 0xA9). */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "\xC3\xA9", 2), FF_UTF8_ACCEPT,
              "utf8 2-byte é");

    /* Valid 3-byte (€ = 0xE2 0x82 0xAC). */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "\xE2\x82\xAC", 3), FF_UTF8_ACCEPT,
              "utf8 3-byte €");

    /* Valid 4-byte (𐍈 = 0xF0 0x90 0x8D 0x88). */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "\xF0\x90\x8D\x88", 4), FF_UTF8_ACCEPT,
              "utf8 4-byte 𐍈");

    /* Standalone continuation byte. */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "\x80", 1), FF_UTF8_REJECT,
              "utf8 standalone 0x80 -> REJECT");

    /* Overlong / illegal lead 0xC0. */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "\xC0\x80", 2), FF_UTF8_REJECT,
              "utf8 overlong 0xC0 0x80 -> REJECT");

    /* Surrogate U+D800 (lead 0xED followed by 0xA0..0xBF) is illegal. */
    EXPECT_EQ(feed(FF_UTF8_ACCEPT, "\xED\xA0\x80", 3), FF_UTF8_REJECT,
              "utf8 surrogate U+D800 -> REJECT");

    /* Truncated sequence: state stays mid-sequence (not ACCEPT, not REJECT). */
    uint32_t s = ff_utf8_decode(FF_UTF8_ACCEPT, 0xE2);  /* expect 2 more bytes */
    if (s == FF_UTF8_ACCEPT || s == FF_UTF8_REJECT) {
        fprintf(stderr, "FAIL: utf8 truncated: state=%u (expected mid-seq)\n", s);
        fails++;
    } else {
        printf("ok: utf8 truncated stays mid-sequence\n");
    }

    /* Fragmented validation: feed the same valid sequence one byte per call. */
    s = FF_UTF8_ACCEPT;
    s = ff_utf8_decode(s, 0xE2);
    s = ff_utf8_decode(s, 0x82);
    s = ff_utf8_decode(s, 0xAC);
    EXPECT_EQ(s, FF_UTF8_ACCEPT, "utf8 fragmented byte-by-byte -> ACCEPT");
}
```

Replace with:

```c
static int read_one_frame(int fd, FfwsFrame *meta, FrameBuf *payload_out)
{
    uint8_t hdr[2];
    int rc = read_n(fd, hdr, 2);
    if (rc != FFE_OK) return rc;

    /* RSV1..3 must be zero unless an extension is negotiated. We don't
     * negotiate any, so any RSV bit set is a protocol error. */
    if (hdr[0] & 0x70) return FFE_PROTOCOL;

    meta->fin    = (hdr[0] & 0x80) ? 1 : 0;
    meta->opcode = hdr[0] & 0x0F;
    meta->masked = (hdr[1] & 0x80) ? 1 : 0;
    uint64_t plen = hdr[1] & 0x7F;

    /* Control frames (opcode >= 0x8) must have FIN=1 and payload <= 125. */
    if ((meta->opcode & 0x8) != 0) {
        if (!meta->fin)   return FFE_PROTOCOL;
        if (plen > 125)   return FFE_PROTOCOL;
    }
```

(All `return FFE_PROTOCOL` sites here will eventually be wrapped via
`ws_close_with` in Task 6; for now the bare return is fine — caller will
still close the fd.)

## Task 7: Rewrite `ws_read_frame` for fragmentation + UTF-8 + clean Close

**Files:**
- Modify: `src/libffproto/ws/ws_module.c`

This is the central change. Two pieces: (a) UTF-8 validation must hook in
where payload bytes land — so we add a thin wrapper around `read_one_frame`
that, for TEXT/CONT data frames in a TEXT message, validates the newly
appended bytes before returning. (b) The state machine in `ws_read_frame`
gains fragmentation handling.

- [ ] **Step 7.1: Replace the existing `ws_read_frame` definition**

Find the existing `ws_read_frame` function:

```c
static int ws_read_frame(int fd, void *ctx, FrameBuf *out)
{
    WsConn *c = (WsConn *)ctx;

    /* Loop until a data frame arrives. Control frames are handled inline. */
    for (;;) {
        FfwsFrame meta;
        size_t mark = out->len;

        int rc = read_one_frame(fd, &meta, out);
        if (rc != FFE_OK) return rc;

        switch (meta.opcode) {
        case FFWS_OP_CLOSE:
            return FFE_CLOSED;

        case FFWS_OP_PING: {
            int wrc = write_frame_raw(fd, FFWS_OP_PONG,
                                      out->data + mark, out->len - mark);
            out->len = mark;
            if (wrc != FFE_OK) return wrc;
            continue;
        }

        case FFWS_OP_PONG:
            out->len = mark;
            continue;

        case FFWS_OP_TEXT:
        case FFWS_OP_BINARY:
            if (!meta.fin) return FFE_PROTOCOL;   /* slice 1: no fragments */
            if (c) c->last_opcode = meta.opcode;
            return FFE_OK;

        default:
            return FFE_PROTOCOL;
        }
    }
}
```

Replace the entire function with:

```c
/* Validate freshly-appended TEXT bytes against the WsConn's incremental
 * UTF-8 state. Returns 0 on OK, FFE_PROTOCOL on REJECT. */
static int validate_text_chunk(WsConn *c, const uint8_t *data, size_t n)
{
    if (!c) return FFE_OK;             /* no ctx → no validation */
    for (size_t i = 0; i < n; i++) {
        c->utf8_state = ff_utf8_decode(c->utf8_state, data[i]);
        if (c->utf8_state == FF_UTF8_REJECT) return FFE_PROTOCOL;
    }
    return FFE_OK;
}

static int ws_read_frame(int fd, void *ctx, FrameBuf *out)
{
    WsConn *c = (WsConn *)ctx;

    /* Loop until a complete data message has been assembled (FIN=1).
     * Control frames are handled inline and don't advance the loop's
     * "did we finish a message" decision. */
    for (;;) {
        FfwsFrame meta;
        size_t mark = out->len;

        int rc = read_one_frame(fd, &meta, out);
        if (rc != FFE_OK) {
            if (rc == FFE_PROTOCOL) ws_close_with(fd, 1002);
            return rc;
        }

        /* --- control frames (don't touch fragment state) --- */
        switch (meta.opcode) {
        case FFWS_OP_CLOSE:
            return FFE_CLOSED;

        case FFWS_OP_PING: {
            int wrc = write_frame_raw(fd, FFWS_OP_PONG,
                                      out->data + mark, out->len - mark);
            out->len = mark;
            if (wrc != FFE_OK) return wrc;
            continue;
        }

        case FFWS_OP_PONG:
            out->len = mark;
            continue;
        }

        /* --- data frames --- */
        switch (meta.opcode) {
        case FFWS_OP_TEXT:
        case FFWS_OP_BINARY:
            /* New data message begins. Must not be mid-fragment. */
            if (c && c->fragment_opcode != 0) {
                ws_close_with(fd, 1002);
                return FFE_PROTOCOL;
            }
            /* For TEXT, reset the UTF-8 validator and validate the new bytes. */
            if (meta.opcode == FFWS_OP_TEXT) {
                if (c) c->utf8_state = FF_UTF8_ACCEPT;
                if (validate_text_chunk(c, out->data + mark, out->len - mark) != FFE_OK) {
                    ws_close_with(fd, 1007);
                    return FFE_PROTOCOL;
                }
            }
            if (meta.fin) {
                /* Single-frame message. */
                if (meta.opcode == FFWS_OP_TEXT &&
                    c && c->utf8_state != FF_UTF8_ACCEPT) {
                    ws_close_with(fd, 1007);
                    return FFE_PROTOCOL;
                }
                if (c) c->last_opcode = meta.opcode;
                return FFE_OK;
            }
            /* Fragment start; remember opcode and keep reading. */
            if (c) c->fragment_opcode = meta.opcode;
            continue;

        case FFWS_OP_CONT:
            /* Continuation requires an active fragment. */
            if (!c || c->fragment_opcode == 0) {
                ws_close_with(fd, 1002);
                return FFE_PROTOCOL;
            }
            if (c->fragment_opcode == FFWS_OP_TEXT) {
                if (validate_text_chunk(c, out->data + mark, out->len - mark) != FFE_OK) {
                    ws_close_with(fd, 1007);
                    return FFE_PROTOCOL;
                }
            }
            if (meta.fin) {
                /* Message complete. */
                if (c->fragment_opcode == FFWS_OP_TEXT &&
                    c->utf8_state != FF_UTF8_ACCEPT) {
                    ws_close_with(fd, 1007);
                    return FFE_PROTOCOL;
                }
                c->last_opcode = c->fragment_opcode;
                c->fragment_opcode = 0;
                return FFE_OK;
            }
            continue;

        default:
            /* Reserved opcodes 0x3..0x7 and 0xB..0xF — clean Close. */
            ws_close_with(fd, 1002);
            return FFE_PROTOCOL;
        }
    }
}
```

Notes for the reader:
- Fragmentation accumulates in the caller's FrameBuf — `read_one_frame`
  appends to `out` regardless of FIN, so by the time we return FFE_OK
  the full message is at `out->data[0..out->len-1]`.
- UTF-8 validation happens *as bytes arrive* (per-frame) and at end-of-
  message. A 4-byte UTF-8 sequence split across two frames passes because
  the DFA state persists in `WsConn.utf8_state`.
- Control frames in the middle of a fragmented message are handled but
  do *not* touch `mark` semantics for the message itself. (`mark` is only
  used to scope the control frame's payload for the pong echo.)
- The `ws_close_with` calls are best-effort; if the write fails the
  caller still gets FFE_PROTOCOL and tears down the connection.
