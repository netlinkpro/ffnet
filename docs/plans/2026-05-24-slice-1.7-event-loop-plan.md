> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:executing-plans (inline) or superpowers:subagent-driven-development to implement task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Migrate `ws-echo` to a non-blocking epoll event loop with multi-worker pthread support via SO_REUSEPORT, evolving the `Protocol` ABI from v2 (blocking call-style) to v3 (event-driven `on_event` callback). CLI subcommands collapse to "parse args + call server_run." See [spec](../specs/2026-05-24-slice-1.7-event-loop-design.md).

**Tech Stack:** C11, NASM x86_64 (mostly unchanged), libc, pthread, epoll, Make, Docker (for Autobahn re-run).

**Commit cadence:** Per project convention (see `feedback-commit-cadence` memory) — **one bundled commit at the end** containing spec + plan + implementation + Autobahn results + memory update. No intermediate commits.

Create `src/libffnet/eventloop.c`:

```c
// SPDX-License-Identifier: LGPL-2.1-or-later
#include "ffnet/ffnet.h"
#include "ffnet/ffutil.h"

#include <errno.h>
#include <stdlib.h>
#include <string.h>
#include <sys/epoll.h>
#include <unistd.h>

typedef struct FdSlot {
    FfEvCallback cb;
    void        *ctx;
    uint32_t     events;        /* FF_EV_* mask */
    int          in_use;
} FdSlot;

struct FfEvLoop {
    int        epfd;
    int        stop_flag;
    FdSlot    *slots;
    size_t     cap;             /* sparse array indexed by fd */
};

static int grow_slots(FfEvLoop *L, int fd)
{
    if ((size_t)fd < L->cap) return FFE_OK;
    size_t new_cap = L->cap ? L->cap : 64;
    while (new_cap <= (size_t)fd) new_cap *= 2;
    FdSlot *p = (FdSlot *)realloc(L->slots, new_cap * sizeof(FdSlot));
    if (!p) return FFE_NOMEM;
    memset(p + L->cap, 0, (new_cap - L->cap) * sizeof(FdSlot));
    L->slots = p;
    L->cap   = new_cap;
    return FFE_OK;
}

static uint32_t to_epoll(uint32_t ev)
{
    uint32_t r = 0;
    if (ev & FF_EV_READ)  r |= EPOLLIN;
    if (ev & FF_EV_WRITE) r |= EPOLLOUT;
    return r;
}

static uint32_t from_epoll(uint32_t ep)
{
    uint32_t r = 0;
    if (ep & (EPOLLIN | EPOLLHUP | EPOLLERR))  r |= FF_EV_READ;
    if (ep & EPOLLOUT)                          r |= FF_EV_WRITE;
    return r;
}

FfEvLoop *ffnet_evloop_new(void)
{
    FfEvLoop *L = (FfEvLoop *)calloc(1, sizeof(FfEvLoop));
    if (!L) return NULL;
    L->epfd = epoll_create1(EPOLL_CLOEXEC);
    if (L->epfd < 0) {
        free(L);
        return NULL;
    }
    return L;
}

void ffnet_evloop_free(FfEvLoop *L)
{
    if (!L) return;
    if (L->epfd >= 0) close(L->epfd);
    free(L->slots);
    free(L);
}

int ffnet_evloop_add(FfEvLoop *L, int fd, uint32_t events,
                     FfEvCallback cb, void *ctx)
{
    if (!L || fd < 0 || !cb) return FFE_INVAL;
    int rc = grow_slots(L, fd);
    if (rc != FFE_OK) return rc;
    if (L->slots[fd].in_use) return FFE_INVAL;

    struct epoll_event ev = { .events = to_epoll(events), .data.fd = fd };
    if (epoll_ctl(L->epfd, EPOLL_CTL_ADD, fd, &ev) < 0) {
        return ffutil_map_errno(-errno);
    }
    L->slots[fd].cb     = cb;
    L->slots[fd].ctx    = ctx;
    L->slots[fd].events = events;
    L->slots[fd].in_use = 1;
    return FFE_OK;
}

int ffnet_evloop_mod(FfEvLoop *L, int fd, uint32_t events)
{
    if (!L || fd < 0 || (size_t)fd >= L->cap || !L->slots[fd].in_use)
        return FFE_INVAL;
    struct epoll_event ev = { .events = to_epoll(events), .data.fd = fd };
    if (epoll_ctl(L->epfd, EPOLL_CTL_MOD, fd, &ev) < 0) {
        return ffutil_map_errno(-errno);
    }
    L->slots[fd].events = events;
    return FFE_OK;
}

int ffnet_evloop_del(FfEvLoop *L, int fd)
{
    if (!L || fd < 0 || (size_t)fd >= L->cap || !L->slots[fd].in_use)
        return FFE_INVAL;
    if (epoll_ctl(L->epfd, EPOLL_CTL_DEL, fd, NULL) < 0) {
        return ffutil_map_errno(-errno);
    }
    L->slots[fd].in_use = 0;
    L->slots[fd].cb     = NULL;
    L->slots[fd].ctx    = NULL;
    L->slots[fd].events = 0;
    return FFE_OK;
}

int ffnet_evloop_run(FfEvLoop *L)
{
    if (!L) return FFE_INVAL;
    struct epoll_event evs[64];
    L->stop_flag = 0;
    for (;;) {
        if (L->stop_flag) return FFE_OK;

        int n = epoll_wait(L->epfd, evs, 64, -1);
        if (n < 0) {
            if (errno == EINTR) continue;
            return ffutil_map_errno(-errno);
        }
        for (int i = 0; i < n; i++) {
            int fd = evs[i].data.fd;
            if ((size_t)fd >= L->cap || !L->slots[fd].in_use) continue;
            uint32_t fired = from_epoll(evs[i].events);
            if (fired == 0) continue;
            L->slots[fd].cb(L, fd, fired, L->slots[fd].ctx);
            /* callback may have called del; loop continues either way */
        }
    }
}

void ffnet_evloop_stop(FfEvLoop *L)
{
    if (L) L->stop_flag = 1;
}
```

// SPDX-License-Identifier: LGPL-2.1-or-later
#include "ffnet/ffnet.h"
#include "ffnet/ffutil.h"

#include <errno.h>
#include <stdio.h>
#include <string.h>
#include <sys/socket.h>
#include <unistd.h>

static int fails = 0;
static int reads_fired = 0;
static int writes_fired = 0;

static void rd_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx)
{
    (void)ctx;
    if (events & FF_EV_READ) {
        char buf[32];
        ssize_t n = read(fd, buf, sizeof buf);
        (void)n;
        reads_fired++;
    }
    if (reads_fired >= 1) ffnet_evloop_stop(loop);
}

static void wr_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx)
{
    (void)loop; (void)fd; (void)ctx;
    if (events & FF_EV_WRITE) writes_fired++;
}

int main(void)
{
    int sv[2];
    if (socketpair(AF_UNIX, SOCK_STREAM, 0, sv) != 0) {
        perror("socketpair"); return 1;
    }

    FfEvLoop *L = ffnet_evloop_new();
    if (!L) { fprintf(stderr, "FAIL: evloop_new\n"); return 1; }
    printf("ok: evloop_new\n");

    /* Register read end. Write a byte from the other end; loop should fire. */
    int rc = ffnet_evloop_add(L, sv[0], FF_EV_READ, rd_cb, NULL);
    if (rc != FFE_OK) { fprintf(stderr, "FAIL: add: %s\n", ffutil_strerror(rc)); fails++; }
    else printf("ok: evloop_add\n");

    write(sv[1], "x", 1);

    rc = ffnet_evloop_run(L);
    if (rc != FFE_OK) { fprintf(stderr, "FAIL: run: %s\n", ffutil_strerror(rc)); fails++; }
    else if (reads_fired != 1) { fprintf(stderr, "FAIL: reads_fired=%d (want 1)\n", reads_fired); fails++; }
    else printf("ok: callback fired once\n");

    /* mod to add WRITE; should fire because sv[0] is writable. */
    rc = ffnet_evloop_mod(L, sv[0], FF_EV_READ | FF_EV_WRITE);
    if (rc != FFE_OK) { fprintf(stderr, "FAIL: mod: %s\n", ffutil_strerror(rc)); fails++; }
    else printf("ok: evloop_mod\n");

    /* del then re-run with no fds should return immediately. */
    rc = ffnet_evloop_del(L, sv[0]);
    if (rc != FFE_OK) { fprintf(stderr, "FAIL: del: %s\n", ffutil_strerror(rc)); fails++; }
    else printf("ok: evloop_del\n");

    ffnet_evloop_free(L);
    printf("ok: evloop_free\n");

    close(sv[0]);
    close(sv[1]);
    (void)wr_cb;   /* silence -Wunused */
    return fails == 0 ? 0 : 1;
}

{
    int flags = fcntl(fd, F_GETFL, 0);
    if (flags < 0) return ffutil_map_errno(-errno);

```c
/* Protocol module ABI v3 (event-driven).
 *
 * Each connection is a state machine owned by the protocol. The framework
 * (libffnet/server.c) is responsible for accept/close-fd; the protocol is
 * responsible for everything that flows over fd.
 *
 * Lifecycle:
 *   open      called once after accept. fd is non-blocking.
 *   on_event  called every time epoll fires with the protocol's
 *             requested events. Protocol reads/writes from/to fd
 *             (handling EAGAIN itself) and returns the next event mask.
 *   close     called when on_event returned 0 or negative, before the
 *             framework closes fd. */
typedef struct Protocol {
    const char *name;

    /* Allocate per-connection state. Returns FFE_OK and writes ctx into
     * *ctx_out, or negative on failure (framework closes fd directly). */
    int (*open)(int fd, void **ctx_out);

    /* Returns:
     *   > 0 — new event mask (FF_EV_READ | FF_EV_WRITE) to wait on next.
     *   = 0 — close this connection cleanly.
     *   < 0 — error; framework logs ffutil_strerror_v(rc) and closes. */
    int (*on_event)(int fd, void *ctx, uint32_t events);

    /* Free per-conn state. Framework closes fd after returning. */
    int (*close)(int fd, void *ctx);
} Protocol;
```

The `ffproto_find`, `ffproto_register`, and `extern const Protocol
ffproto_websocket` declarations stay unchanged.

- [ ] **Step 3.3: Implement `src/libffnet/server.c`**

Create `src/libffnet/server.c`:

```c
// SPDX-License-Identifier: LGPL-2.1-or-later
#include "ffnet/ffnet.h"
#include "ffnet/ffproto.h"
#include "ffnet/ffutil.h"

#include <errno.h>
#include <pthread.h>
#include <signal.h>
#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <unistd.h>

/* Per-worker self-pipe for stop. main thread writes; worker reads. */
typedef struct Worker {
    int               id;
    int               stop_pipe[2];      /* [0]=read, [1]=write */
    const Protocol   *proto;
    const char       *host;
    uint16_t          port;
    int               listen_fd;
    FfEvLoop         *loop;
    pthread_t         thread;
} Worker;

/* Per-connection wrapper: holds the protocol's ctx + back-ref to worker for
 * the framework's bookkeeping. Allocated on accept, freed on close. */
typedef struct Conn {
    void   *proto_ctx;
    Worker *w;
} Conn;

static void conn_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx);
static void accept_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx);
static void stop_pipe_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx);

static void conn_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx)
{
    Conn *c = (Conn *)ctx;
    int rc = c->w->proto->on_event(fd, c->proto_ctx, events);
    if (rc > 0) {
        /* New event mask; mod. */
        ffnet_evloop_mod(loop, fd, (uint32_t)rc);
        return;
    }
    if (rc < 0) {
        FF_LOGW("conn fd=%d: %s", fd, ffutil_strerror_v(rc));
    }
    /* Close path (rc == 0 or rc < 0): tear down. */
    c->w->proto->close(fd, c->proto_ctx);
    ffnet_evloop_del(loop, fd);
    close(fd);
    free(c);
}

static void accept_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx)
{
    (void)events;
    Worker *w = (Worker *)ctx;
    /* Drain the accept backlog (LT: keep accepting until EAGAIN). */
    for (;;) {
        int cfd = ffnet_tcp_accept(fd);
        if (cfd == FFE_AGAIN) return;
        if (cfd < 0) {
            FF_LOGW("worker %d accept: %s", w->id, ffutil_strerror_v(cfd));
            return;
        }

        int rc = ffnet_set_nonblocking(cfd);
        if (rc != FFE_OK) {
            FF_LOGW("worker %d set_nonblocking: %s", w->id, ffutil_strerror_v(rc));
            close(cfd);
            continue;
        }

        Conn *c = (Conn *)calloc(1, sizeof(Conn));
        if (!c) { close(cfd); continue; }
        c->w = w;

        rc = w->proto->open(cfd, &c->proto_ctx);
        if (rc != FFE_OK) {
            FF_LOGW("worker %d proto->open: %s", w->id, ffutil_strerror_v(rc));
            free(c);
            close(cfd);
            continue;
        }

        rc = ffnet_evloop_add(loop, cfd, FF_EV_READ, conn_cb, c);
        if (rc != FFE_OK) {
            FF_LOGW("worker %d evloop_add: %s", w->id, ffutil_strerror_v(rc));
            w->proto->close(cfd, c->proto_ctx);
            free(c);
            close(cfd);
            continue;
        }
    }
}

static void stop_pipe_cb(FfEvLoop *loop, int fd, uint32_t events, void *ctx)
{
    (void)events; (void)ctx;
    /* Drain the byte(s) and stop. */
    char buf[16];
    while (read(fd, buf, sizeof buf) > 0) {}
    ffnet_evloop_stop(loop);
}

static void *worker_main(void *arg)
{
    Worker *w = (Worker *)arg;

    w->listen_fd = ffnet_tcp_create();
    if (w->listen_fd < 0) {
        FF_LOGE("worker %d create: %s", w->id, ffutil_strerror_v(w->listen_fd));
        return NULL;
    }
    int rc;
    if ((rc = ffnet_tcp_set_reuseport(w->listen_fd)) != FFE_OK) {
        FF_LOGE("worker %d reuseport: %s", w->id, ffutil_strerror_v(rc));
        goto err;
    }
    if ((rc = ffnet_set_nonblocking(w->listen_fd)) != FFE_OK) {
        FF_LOGE("worker %d nonblocking: %s", w->id, ffutil_strerror_v(rc));
        goto err;
    }
    if ((rc = ffnet_tcp_bind_listen(w->listen_fd, w->host, w->port, 128)) != FFE_OK) {
        FF_LOGE("worker %d bind_listen %s:%u: %s",
                w->id, w->host, (unsigned)w->port, ffutil_strerror_v(rc));
        goto err;
    }

    w->loop = ffnet_evloop_new();
    if (!w->loop) { FF_LOGE("worker %d evloop_new", w->id); goto err; }

    rc = ffnet_evloop_add(w->loop, w->listen_fd, FF_EV_READ, accept_cb, w);
    if (rc != FFE_OK) {
        FF_LOGE("worker %d evloop_add listen: %s", w->id, ffutil_strerror_v(rc));
        goto err_loop;
    }
    rc = ffnet_evloop_add(w->loop, w->stop_pipe[0], FF_EV_READ, stop_pipe_cb, NULL);
    if (rc != FFE_OK) {
        FF_LOGE("worker %d evloop_add stop_pipe: %s", w->id, ffutil_strerror_v(rc));
        goto err_loop;
    }

    if (w->id == 0) {
        int p = ffnet_tcp_local_port(w->listen_fd);
        if (p > 0) {
            printf("listening on %s:%d (workers=%d)\n", w->host, p, 1);
            fflush(stdout);
        }
    }

    ffnet_evloop_run(w->loop);

err_loop:
    if (w->loop) ffnet_evloop_free(w->loop);
err:
    if (w->listen_fd >= 0) close(w->listen_fd);
    return NULL;
}

/* --- signal handler shim --- */
static Worker *g_workers = NULL;
static int     g_n_workers = 0;

static void on_sigint(int signo)
{
    (void)signo;
    for (int i = 0; i < g_n_workers; i++) {
        if (g_workers[i].stop_pipe[1] >= 0)
            (void)!write(g_workers[i].stop_pipe[1], "x", 1);
    }
}

int ffnet_server_run(const Protocol *proto, const char *host,
                     uint16_t port, int workers)
{
    if (!proto || !host || workers < 1) return FFE_INVAL;

    Worker *ws = (Worker *)calloc((size_t)workers, sizeof(Worker));
    if (!ws) return FFE_NOMEM;
    for (int i = 0; i < workers; i++) {
        ws[i].id        = i;
        ws[i].proto     = proto;
        ws[i].host      = host;
        ws[i].port      = port;
        ws[i].listen_fd = -1;
        ws[i].stop_pipe[0] = -1;
        ws[i].stop_pipe[1] = -1;
        if (pipe(ws[i].stop_pipe) != 0) {
            free(ws);
            return ffutil_map_errno(-errno);
        }
        ffnet_set_nonblocking(ws[i].stop_pipe[0]);
    }

    g_workers   = ws;
    g_n_workers = workers;
    struct sigaction sa = { .sa_handler = on_sigint };
    sigaction(SIGINT, &sa, NULL);
    sigaction(SIGTERM, &sa, NULL);

    if (workers == 1) {
        worker_main(&ws[0]);
    } else {
        for (int i = 0; i < workers; i++) {
            pthread_create(&ws[i].thread, NULL, worker_main, &ws[i]);
        }
        /* Worker 0 prints the listen line on bind; others stay quiet. We
         * still need a single-process "listening on" line in multi-worker
         * mode; the first worker's worker_main handles it. */
        for (int i = 0; i < workers; i++) {
            pthread_join(ws[i].thread, NULL);
        }
    }

    for (int i = 0; i < workers; i++) {
        if (ws[i].stop_pipe[0] >= 0) close(ws[i].stop_pipe[0]);
        if (ws[i].stop_pipe[1] >= 0) close(ws[i].stop_pipe[1]);
    }
    free(ws);
    g_workers = NULL;
    g_n_workers = 0;
    return FFE_OK;
}
```
